# M1 Brain Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Load the MaleCNS v1.0 brain into a signed sparse graph, run it as a Shiu-style LIF network in JAX, and prove the kernel matches a Brian2 oracle spike for spike.

**Architecture:** `flycraftax/data.py` turns the three MaleCNS feather files into a `Connectome` (neuron table + signed edge arrays, cached as npz). `flycraftax/brain.py` holds the LIF step as a pure JAX function over a `BrainState` with a BCOO weight matrix, batched over environments. `flycraftax/oracle.py` is a Brian2 re-implementation used only in tests. Scripts cover download, throughput benchmark, and the MN9 smoke check.

**Tech Stack:** Python 3.11 in conda env `flycraftax`, JAX with CUDA 12, numpy, pandas, pyarrow, pytest, brian2 (test only).

**Spec:** `tracker.md` (Spec > Brain, Milestones > M1). Reference material: `context/malecns_data.md`, `context/brain_model.md`, `context/doomfly.md`.

## Global Constraints

- Population: all rows with `superclass` set and not starting with `vnc`. No `status` filter. Expect 146,271 neurons.
- Edge file: `connectome-weights-male-cns-v1.0-minconf-0.5-traced-only.feather` (508 MB, columns `body_pre, body_post, weight, type_pre, type_post`, sorted by weight descending).
- Threshold: config value, default `weight >= 5`.
- Sign by `consensus_nt` of the presynaptic neuron: acetylcholine +1; gaba, glutamate, histamine -1; dopamine, serotonin, octopamine +1; `unclear` falls back to `celltype_predicted_nt` with the same table; anything still unsigned gets sign 0 (edge dropped).
- LIF constants (Shiu et al. 2024): v_rest -52 mV, v_reset -52 mV, v_th -45 mV, t_mbr 20 ms, tau 5 ms, t_rfc 2.2 ms, t_dly 1.8 ms, w_syn 0.275 mV, Poisson kick = w_syn x 250 = 68.75 mV, dt default 0.1 ms.
- Exact exponential update, never forward Euler.
- Delay buffer stores spikes, not currents. Silencing is a spike mask. State persists across windows.
- License MIT. Port from `external/fly-brain/code/paper-phil-drosophila/model.py` (MIT) and the paper only. Do not copy from `external/fly-brain/code/run_pytorch.py` (GPL).
- Style: minimal code, no speculative config, no flaky tests. Tests that need the full data are marked `slow`.

```python
        if dest.exists():
```

```python
@dataclass
class Connectome:
    body_id: np.ndarray   # int64 (N), sorted ascending, index = neuron id used everywhere else
    type: np.ndarray      # object (N), cell type string or ""
    cls: np.ndarray       # object (N), `class` column or ""
    superclass: np.ndarray  # object (N)
    side: np.ndarray      # object (N), "L" / "R" / "M" / "unknown"; somaSide with rootSide fallback
    sign: np.ndarray      # float32 (N), +1 / -1 / 0
    pre: np.ndarray       # int32 (E), presynaptic neuron index
    post: np.ndarray      # int32 (E), postsynaptic neuron index
    count: np.ndarray     # int32 (E), synapse count

    @property
    def n(self) -> int
    def index(self, type_prefix: str | None = None, types: list[str] | None = None,
              cls: str | None = None, side: str | None = None) -> np.ndarray  # int32 indices
    def signed_count(self) -> np.ndarray  # float32 (E) = sign[pre] * count

def load_connectome(threshold: int = 5, data_dir: Path = Path("data")) -> Connectome
```

- `index()` combines the given filters with AND. `type_prefix` matches `type.startswith`. `types` matches exact strings. Edges with sign 0 are removed inside `load_connectome`, so `signed_count` is never 0.
- Result cached at `data/connectome_t{threshold}.npz`; rebuilt if missing.

```python
# flycraftax/data.py
"""MaleCNS v1.0 loader: three feather files in, one signed sparse graph out."""
from dataclasses import dataclass
from pathlib import Path

import numpy as np
import pandas as pd

SIGN = {
    "acetylcholine": 1.0,
    "gaba": -1.0,
    "glutamate": -1.0,
    "histamine": -1.0,
    "dopamine": 1.0,
    "serotonin": 1.0,
    "octopamine": 1.0,
}


@dataclass
class Connectome:
    body_id: np.ndarray
    type: np.ndarray
    cls: np.ndarray
    superclass: np.ndarray
    side: np.ndarray
    sign: np.ndarray
    pre: np.ndarray
    post: np.ndarray
    count: np.ndarray

    @property
    def n(self) -> int:
        return len(self.body_id)

    def index(self, type_prefix=None, types=None, cls=None, side=None) -> np.ndarray:
        keep = np.ones(self.n, dtype=bool)
        if type_prefix is not None:
            keep &= np.char.startswith(self.type.astype(str), type_prefix)
        if types is not None:
            keep &= np.isin(self.type, types)
        if cls is not None:
            keep &= self.cls == cls
        if side is not None:
            keep &= self.side == side
        return np.flatnonzero(keep).astype(np.int32)

    def signed_count(self) -> np.ndarray:
        return (self.sign[self.pre] * self.count).astype(np.float32)


def _neurons(data_dir: Path) -> pd.DataFrame:
    ann = pd.read_feather(data_dir / "body-annotations.feather")
    ann = ann[ann.superclass.notna() & ~ann.superclass.str.startswith("vnc")]
    nt = pd.read_feather(
        data_dir / "body-neurotransmitters.feather",
        columns=["body", "consensus_nt", "celltype_predicted_nt"],
    )
    df = ann.merge(nt, left_on="bodyId", right_on="body", how="left")
    sign = df.consensus_nt.map(SIGN)
    fallback = df.celltype_predicted_nt.map(SIGN)
    df["sign"] = sign.fillna(fallback).fillna(0.0).astype(np.float32)
    df["side"] = df.somaSide.fillna(df.rootSide).fillna("unknown")
    return df.sort_values("bodyId").reset_index(drop=True)


def _edges(data_dir: Path, body_id: np.ndarray, threshold: int) -> tuple[np.ndarray, ...]:
    import pyarrow.feather as feather

    tbl = feather.read_table(data_dir / "edges-traced.feather", columns=["body_pre", "body_post", "weight"])
    e = tbl.to_pandas()
    e = e[e.weight >= threshold]
    pre = np.searchsorted(body_id, e.body_pre.values)
    post = np.searchsorted(body_id, e.body_post.values)
    ok = (pre < len(body_id)) & (post < len(body_id))
    ok &= (body_id[np.minimum(pre, len(body_id) - 1)] == e.body_pre.values)
    ok &= (body_id[np.minimum(post, len(body_id) - 1)] == e.body_post.values)
    return pre[ok].astype(np.int32), post[ok].astype(np.int32), e.weight.values[ok].astype(np.int32)


def load_connectome(threshold: int = 5, data_dir: Path = Path("data")) -> Connectome:
    cache = data_dir / f"connectome_t{threshold}.npz"
    if cache.exists():
        z = np.load(cache, allow_pickle=True)
        return Connectome(**{k: z[k] for k in z.files})

    df = _neurons(data_dir)
    body_id = df.bodyId.values.astype(np.int64)
    pre, post, count = _edges(data_dir, body_id, threshold)
    sign = df.sign.values
    keep = sign[pre] != 0
    conn = Connectome(
        body_id=body_id,
        type=df.type.fillna("").values.astype(object),
        cls=df["class"].fillna("").values.astype(object),
        superclass=df.superclass.values.astype(object),
        side=df.side.values.astype(object),
        sign=sign,
        pre=pre[keep],
        post=post[keep],
        count=count[keep],
    )
    np.savez(cache, **conn.__dict__)
    return conn
```

```python
def run_brian2(n: int, pre: np.ndarray, post: np.ndarray, w_mv: np.ndarray,
               kicks: np.ndarray, t_ms: float, p: BrainParams | None = None) -> np.ndarray
```
  - `w_mv` is the signed weight per edge in mV, i.e. `signed_count * w_syn`.
  - `kicks` is a bool array `(T, n)`; where True, `v += p.kick` mV is applied to that neuron at that step, delivered in Brian2's synapse slot (same as Shiu's `PoissonInput`). Neurons with any True in their column get refractory 0, as Shiu does for driven neurons.
  - Returns a bool array `(T, n)` of spikes, `T = round(t_ms / p.dt_ms)`.
- Consumes: `BrainParams` from `flycraftax.brain` (defined in Task 4; for this task create it there with only the constants and no step function).

- [ ] **Step 1: Create `flycraftax/brain.py` with just the params**

```python
# flycraftax/brain.py
"""Shiu et al. 2024 LIF on a sparse connectome, in JAX."""
import math
from typing import NamedTuple


class BrainParams(NamedTuple):
    dt_ms: float = 0.1
    v_rest: float = -52.0
    v_reset: float = -52.0
    v_th: float = -45.0
    t_mbr: float = 20.0
    tau: float = 5.0
    t_rfc: float = 2.2
    t_dly: float = 1.8
    w_syn: float = 0.275
    kick: float = 0.275 * 250

    @property
    def n_dly(self) -> int:
        return round(self.t_dly / self.dt_ms)

    @property
    def n_rfc(self) -> int:
        return round(self.t_rfc / self.dt_ms)

    def decay(self) -> tuple[float, float, float]:
        a = math.exp(-self.dt_ms / self.tau)
        b = math.exp(-self.dt_ms / self.t_mbr)
        c = (b - a) * self.tau / (self.t_mbr - self.tau)
        return a, b, c
```

```python
# flycraftax/oracle.py
"""Brian2 implementation of the Shiu LIF model. Test dependency only."""
import numpy as np

from flycraftax.brain import BrainParams

EQS = """
dv/dt = (v_0 - v + g) / t_mbr : volt (unless refractory)
dg/dt = -g / tau               : volt (unless refractory)
rfc                            : second
"""


def run_brian2(n, pre, post, w_mv, kicks, t_ms, p=None):
    import brian2 as b2

    p = p or BrainParams()
    b2.start_scope()
    b2.defaultclock.dt = p.dt_ms * b2.ms
    ns = dict(v_0=p.v_rest * b2.mV, t_mbr=p.t_mbr * b2.ms, tau=p.tau * b2.ms, v_th=p.v_th * b2.mV)
    neu = b2.NeuronGroup(
        n, EQS, method="linear", threshold="v > v_th",
        reset=f"v = {p.v_reset}*mV; g = 0*mV", refractory="rfc", namespace=ns,
    )
    neu.v = p.v_rest * b2.mV
    neu.g = 0 * b2.mV
    neu.rfc = p.t_rfc * b2.ms
    driven = np.flatnonzero(kicks.any(axis=0))
    neu.rfc[driven] = 0 * b2.ms

    objs = [neu]
    if len(pre):
        syn = b2.Synapses(neu, neu, "w : volt", on_pre="g += w", delay=p.t_dly * b2.ms)
        syn.connect(i=np.asarray(pre), j=np.asarray(post))
        syn.w = np.asarray(w_mv) * b2.mV
        objs.append(syn)

    t_idx, n_idx = np.nonzero(kicks)
    if len(t_idx):
        gen = b2.SpikeGeneratorGroup(n, n_idx, t_idx * p.dt_ms * b2.ms)
        kick = b2.Synapses(gen, neu, on_pre=f"v += {p.kick}*mV")
        kick.connect(j="i")
        objs += [gen, kick]

    mon = b2.SpikeMonitor(neu)
    net = b2.Network(*objs, mon)
    net.run(t_ms * b2.ms)

    T = round(t_ms / p.dt_ms)
    out = np.zeros((T, n), dtype=bool)
    steps = np.round(mon.t / b2.defaultclock.dt).astype(int)
    out[steps[steps < T], mon.i[steps < T]] = True
    return out
```

```python
class BrainState(NamedTuple):
    v: jax.Array        # (B, N) float32
    g: jax.Array        # (B, N) float32
    refrac: jax.Array   # (B, N) int32, steps remaining
    buf: jax.Array      # (n_dly, B, N) float32, buf[0] is the oldest spike vector
    counts: jax.Array   # (B, N) float32, spikes accumulated since last reset_counts

def build_weights(n: int, pre, post, signed_count, p: BrainParams) -> BCOO   # (N, N), W[post, pre] = signed_count * w_syn, mV
def rfc_steps(n: int, driven_idx, p: BrainParams) -> jax.Array                # (N,) int32, 0 for driven neurons
def init_state(n: int, batch: int, p: BrainParams) -> BrainState
def step(W, rfc, p, state, kick_prob, silence, key) -> tuple[BrainState, jax.Array]   # kick_prob (B, N) float32 per-step Bernoulli prob; silence (N,) float32 mask 1 = active; returns spikes (B, N) bool
def run_window(W, rfc, p, state, kick_prob, silence, key, n_steps: int) -> BrainState   # lax.scan over step, counts accumulated
def rate_hz(counts, n_steps: int, p: BrainParams) -> jax.Array
def reset_counts(state) -> BrainState
def rate_for_hz(hz, p) -> float   # per-step Bernoulli probability = hz * dt_ms / 1000
```

- Step order must reproduce Brian2: (1) exact-exponential state update for non-refractory neurons, (2) threshold, (3) delayed synaptic input added to `g` and kicks added to `v`, (4) reset spiking neurons, decrement refractory counters, push spikes into the buffer. Whether synaptic input lands on refractory neurons is decided by the oracle in Step 3 below.
- Kick is applied as `v += kick` exactly like Shiu, not as a forced spike, so the two implementations are step-for-step comparable.

```python
import jax
import jax.numpy as jnp
import numpy as np
from jax.experimental import sparse


class BrainState(NamedTuple):
    v: jax.Array
    g: jax.Array
    refrac: jax.Array
    buf: jax.Array
    counts: jax.Array


def build_weights(n, pre, post, signed_count, p):
    idx = jnp.stack([jnp.asarray(post, jnp.int32), jnp.asarray(pre, jnp.int32)], axis=1)
    vals = jnp.asarray(signed_count, jnp.float32) * p.w_syn
    return sparse.BCOO((vals, idx), shape=(n, n)).sort_indices()


def rfc_steps(n, driven_idx, p):
    return jnp.full(n, p.n_rfc, jnp.int32).at[jnp.asarray(driven_idx, jnp.int32)].set(0)


def init_state(n, batch, p):
    z = jnp.zeros((batch, n), jnp.float32)
    return BrainState(
        v=jnp.full((batch, n), p.v_rest, jnp.float32),
        g=z, refrac=jnp.zeros((batch, n), jnp.int32),
        buf=jnp.zeros((p.n_dly, batch, n), jnp.float32), counts=z,
    )


def rate_for_hz(hz, p):
    return hz * p.dt_ms / 1000.0


def step(W, rfc, p, state, kick_prob, silence, key):
    a, b, c = p.decay()
    active = state.refrac == 0
    v = jnp.where(active, p.v_rest + b * (state.v - p.v_rest) + c * state.g, state.v)
    g = jnp.where(active, a * state.g, state.g)
    spikes = v > p.v_th
    delayed = state.buf[0] * silence                      # (B, N), spikes from n_dly steps ago
    syn_in = (W @ delayed.T).T                            # (N, N) @ (N, B) -> (B, N)
    g = g + syn_in                                        # adjust per Step 3 finding if Brian2 drops input on refractory neurons
    v = v + p.kick * jax.random.bernoulli(key, kick_prob)
    v = jnp.where(spikes, p.v_reset, v)
    g = jnp.where(spikes, 0.0, g)
    refrac = jnp.where(spikes, rfc[None, :], jnp.maximum(state.refrac - 1, 0))
    sp = spikes.astype(jnp.float32)
    buf = jnp.concatenate([state.buf[1:], sp[None]], axis=0)
    return BrainState(v, g, refrac, buf, state.counts + sp), spikes


def run_window(W, rfc, p, state, kick_prob, silence, key, n_steps):
    def body(carry, k):
        st, _ = step(W, rfc, p, carry, kick_prob, silence, k)
        return st, None
    keys = jax.random.split(key, n_steps)
    state, _ = jax.lax.scan(body, state, keys)
    return state


def rate_hz(counts, n_steps, p):
    return counts / (n_steps * p.dt_ms / 1000.0)


def reset_counts(state):
    return state._replace(counts=jnp.zeros_like(state.counts))
```

```python
def subgraph(conn, seed_idx, hops=2, max_n=2000):
    mag = np.abs(conn.signed_count())
    selected = list(np.asarray(seed_idx, np.int32))
    chosen = set(selected)
    frontier = np.array(selected)
    for _ in range(hops):
        m = np.isin(conn.pre, frontier)
        score = np.bincount(conn.post[m], weights=mag[m], minlength=conn.n)
        cand = [i for i in np.argsort(-score) if score[i] > 0 and i not in chosen]
        cand = cand[: max_n - len(selected)]
        selected += cand
        chosen.update(cand)
        frontier = np.array(cand)
        if len(selected) >= max_n:
            break
    nodes = np.array(selected, np.int32)
    remap = np.full(conn.n, -1, np.int32)
    remap[nodes] = np.arange(len(nodes), dtype=np.int32)
    keep = (remap[conn.pre] >= 0) & (remap[conn.post] >= 0)
    return nodes, remap[conn.pre[keep]], remap[conn.post[keep]], conn.signed_count()[keep]
```
